cmake_minimum_required(VERSION 3.14)
project(edisk_installer_native LANGUAGES CXX)
add_library(edisk_installation STATIC installation.cpp)
target_compile_features(edisk_installation PUBLIC cxx_std_17)
target_compile_definitions(edisk_installation PUBLIC UNICODE _UNICODE NOMINMAX _WIN32_WINNT=0x0A00)
target_compile_options(edisk_installation PRIVATE /W4 /WX /EHsc /utf-8)
target_link_libraries(edisk_installation PUBLIC bcrypt)
set_property(TARGET edisk_installation PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(edisk-launcher WIN32 launcher.cpp)
target_link_libraries(edisk-launcher PRIVATE edisk_installation)
set_property(TARGET edisk-launcher PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

add_executable(guard_harness guard_harness.cpp)
target_link_libraries(guard_harness PRIVATE edisk_installation)
set_property(TARGET guard_harness PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

add_executable(fixture_runner WIN32 fixture_runner.cpp)
target_link_libraries(fixture_runner PRIVATE edisk_installation)
set_property(TARGET fixture_runner PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

add_executable(edisk-launcher-test WIN32 launcher.cpp)
target_compile_definitions(edisk-launcher-test PRIVATE EDISK_COORD_TEST)
target_link_libraries(edisk-launcher-test PRIVATE edisk_installation)
set_property(TARGET edisk-launcher-test PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
